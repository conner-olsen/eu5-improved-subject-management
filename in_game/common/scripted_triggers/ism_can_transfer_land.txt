# Can a location be transferred from one country to another
# ROOT: location
# ARGS: $giver$:country, $receiver$:country
ism_can_transfer_location = {
    owner ?= $giver$
    OR = {
        # Can always take land from colonial nations
        AND = {
            $giver$ = { is_subject_type = colonial_nation }
            $receiver$ = scope:actor
        }
        OR = {
            any_coast_border_location = {
                owner ?= $receiver$
            }
            sub_continent = $receiver$.capital.sub_continent
        }
    }
}

# Can a province be transferred from one country to another
# ROOT: province
# ARGS: $giver$:country, $receiver$:country
ism_can_transfer_province = {
    any_location_in_province = {
        ism_can_transfer_location = {
            giver = $giver$
            receiver = $receiver$
        }
    }
}

# Can the $giver$ transfer any land to the $receiver$
# ROOT: country interaction
# ARGS: $giver$:country, $receiver$:country
ism_can_transfer_land = {
    NOT = { $receiver$ = $giver$ }
    $receiver$ = {
        NOR = {
            country_type = building
            country_type = pop
        }
    }
    $giver$ = {
        NOR = {
            country_type = building
            country_type = pop
        }
    }
    $giver$ = {
        any_owned_location = {
            ism_can_transfer_location = {
                giver = $giver$
                receiver = $receiver$
            }
        }
    }
}

# Does the Maintain Vanilla Balance rule cover this subject
# ROOT: any
# ARGS: $giver$:country
ism_vanilla_balance_protects = {
    has_global_variable = ism__maintain_vanilla_balance
    NOT = { $giver$ = scope:actor }
    $giver$ = {
        is_subject = yes
        is_colonial_subject = no
        NOT = {
            AND = {
                exists = var:ism_created_by
                var:ism_created_by = scope:actor
            }
        }
    }
}

# Will the giver still hold land once one more location leaves
# ROOT: any
ism_giver_keeps_a_location = {
    OR = {
        NOT = { ism_vanilla_balance_protects = { giver = scope:actor.var:ism_giver_nation } }
        scope:actor.var:ism_giver_nation = { num_locations > 1 }
    }
}

# Decides if the $giver$ is allowed to transfer land to the $receiver$
# Differs from ism_can_transfer_land which only checks if a land transfer is possible.
# This goes in the `enabled` vs `visible` check.
# ROOT: country interaction
# ARGS: $giver$:country, $receiver$:country
ism_allow_transfer_land = {
    # Cannot take land from non-colonial subjects
    OR = {
        NOT = { scope:actor = $receiver$ }
        $giver$ = { is_subject_type = colonial_nation }
    }
    # Cannot transfer land from non-colonial subjects to colonial subjects
    # To prevent transferring from vassal->colonial subject->self.
    OR = {
        NOT = { $giver$ = { is_subject_type = colonial_nation } }
        $receiver$ = { is_subject_type = colonial_nation }
        scope:actor = $receiver$
    }
    OR = {
        # Can always transfer from colonial nations
        $giver$ = { is_subject_type = colonial_nation }
        # Can always transfer land to colonial nation from yourself
        AND = {
            $receiver$ = { is_subject_type = colonial_nation }
            scope:actor = $giver$
        }
        # Cannot transfer land from a nation to a nation if they are not in the same war
        custom_tooltip = {
            text = "is_in_same_war"
            OR = {
                $giver$ = { at_war = no }
                AND = {
                    $giver$ = { at_war = yes }
                    $receiver$ = { at_war = yes }
                }
            }
        }
    }
    custom_tooltip = {
        text = "ism_subject_must_keep_a_location"
        OR = {
            NOT = { ism_vanilla_balance_protects = { giver = $giver$ } }
            $giver$ = { num_locations > 1 }
        }
    }
}

ism_allow_transfer_location = {
    OR = {
        NOT = { scope:actor.capital = this }
        NOT = { scope:actor = scope:actor.var:ism_giver_nation }
    }
}
