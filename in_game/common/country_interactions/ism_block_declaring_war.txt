ism_block_subject_declaring_war = {
	type = subject
	category = CATEGORY_SUBJECT_ACTIONS
	show_message = no
	show_message_to_target = no

	potential = {
		scope:actor = {
			NOT = { has_variable = ism__block_declaring_war }
		}
	}

	allow = {
	}

	ai_tick = never

	use_enroute = no

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			scope:actor = {
				every_subject = {
					add_to_list = source
				}
			}
		}
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			NOR = {
				has_country_modifier = ism_overlord_blocking_war_declarations
				has_variable = ism__suspended_war_declaration_block
			}
			NOT = { modifier:blocked_from_declaring_war = yes }
		}
		enabled = {
			subject_loyalty >= MIN_SUBJECT_LOYALTY_FOR_ACTION
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:recipient
			}
			scope:recipient = {
				add_country_modifier = {
					modifier = ism_overlord_blocking_war_declarations
					days = -1
					mode = replace
				}
			}
		}
	}
}
