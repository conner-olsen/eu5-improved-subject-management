INJECT:intervene_in_subject_regular_war = {
	select_trigger = {
		looking_for_a = country
		interaction_source_list = { }
		target_flag = recipient
		name = "choose_country"
		show_if = {
			scope:actor = { is_human = yes }
		}
		visible = {
			always = no
		}
	}
}

# No point in this appearing for colonial subjects.
INJECT:seize_location_from_subject = {
	select_trigger = {
		looking_for_a = country
		interaction_source_list = { }
		target_flag = recipient
		name = "choose_country"
		show_if = {
			exists = scope:recipient
			scope:recipient = { is_colonial_subject = yes }
			scope:actor = { is_human = yes }
		}
		visible = {
			always = no
		}
	}
}
