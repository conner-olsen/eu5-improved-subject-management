ism_take_over_subject_wars = {
	$selected_subject$ = {
		every_current_war = {
			if = {
				limit = {
					$selected_subject$ = { is_war_leader_of = prev }
					NOT = { is_in_war = $selected_overlord$ }
				}
				if = {
					limit = {
						is_civil_war_for = $selected_subject$
					}
					$selected_overlord$ = {
						reverse_add_opinion = {
							target = $selected_subject$
							modifier = opinion_helped_in_civil_war
						}
					}
				}
				if = {
					limit = {
						attacker_leader = $selected_subject$
					}
					$selected_overlord$ = {
						join_war_as_attacker = {
							war = prev
							reason = intervene
							ignore_rules = yes
						}
					}
				}
				else = {
					$selected_overlord$ = {
						join_war_as_defender = {
							war = prev
							reason = intervene
							ignore_rules = yes
						}
					}
				}
			}
		}
	}
	$selected_overlord$ = {
		take_over_all_wars = $selected_subject$
	}
	# Attempt to fix strange vanilla bug, #TODO Verify if working
	$selected_overlord$ = {
		if = {
			limit = { is_rival_of = $selected_subject$ }
			$selected_subject$ = { remove_rival = $selected_overlord$ }
		}
	}
	$selected_subject$ = {
		if = {
			limit = { is_rival_of = $selected_overlord$ }
			$selected_overlord$ = { remove_rival = $selected_subject$ }
		}
	}
}

ism_block_subject_war_declarations = {
	if = {
		limit = { is_overlord = yes }
		save_scope_as = overlord_scope
		every_subject = {
			if = {
				limit = {
					scope:overlord_scope = { has_variable = ism__block_declaring_war }
					NOT = { has_country_modifier = ism_overlord_blocking_all_subject_war_declarations }
				}
				add_country_modifier = {
					modifier = ism_overlord_blocking_all_subject_war_declarations
					days = -1
					mode = replace
				}
			}
			else = {
				if = {
					limit = {
						has_country_modifier = ism_overlord_blocking_all_subject_war_declarations
					}
					remove_country_modifier = ism_overlord_blocking_all_subject_war_declarations
				}
			}
		}
	}
}
